    50	def test_start_delay_p95_under_normal_load(engine: Engine) -> None:
    51	    """V-P5-27: p95 of (task created minus scheduled_for) ≤ 60 s at 20 due Runs per minute."""
    52	    fx = Fixture.create(engine, f"lat{uuid.uuid4().hex[:6]}", FixedClock(T0))
    53	    delays: list[float] = []
    54	    with Session(engine) as s, s.begin():
    55	        for i in range(SCHEDULES):
    56	            # no Agent selection: the simulation measures scheduler latency, not routing
    57	            fx.schedule(s, f"sch-lat-{i:03d}", concurrency="ALLOW", agent_selection={})
    58	
    59	    minute = 0
    60	    created = 0
    61	    with Session(engine) as s, s.begin():
    62	        pending: list[tuple[str, dt.datetime]] = []
    63	        while created < SCHEDULES:
    64	            # a minute's worth of occurrences becomes due at the top of the minute
    65	            due_at = T0 + dt.timedelta(minutes=minute)
    66	            for _ in range(DUE_PER_MINUTE):
    67	                if created >= SCHEDULES:
    68	                    break
    69	                run_id = f"run-lat-{created:03d}"
    70	                set_clock(fx.clock, due_at)
    71	                fx.run(
    72	                    s,
    73	                    f"sch-lat-{created:03d}",
    74	                    run_id=run_id,
    75	                    status="DUE",
    76	                    scheduled_for=due_at,
    77	                )
    78	                pending.append((run_id, due_at))
    79	                created += 1
    80	            # four ticks per minute, two runners each claiming a bounded batch
    81	            for tick in range(4):
    82	                set_clock(fx.clock, due_at + dt.timedelta(seconds=TICK_S * (tick + 1)))
    83	                for _ in range(RUNNERS):
    84	                    batch, pending = (
    85	                        pending[:BATCH_PER_RUNNER_TICK],
    86	                        pending[BATCH_PER_RUNNER_TICK:],
    87	                    )
    88	                    for run_id, scheduled_for in batch:
    89	                        ctx = fx.ctx(s)
    90	                        s.execute(
    91	                            text(
    92	                                "UPDATE schedule_runs SET status = 'CLAIMED', claimed_by = 'r', "
    93	                                "claimed_at = :now, lease_expires_at = :lease WHERE run_id = :r"
    94	                            ),
    95	                            {
    96	                                "now": ctx.now,
    97	                                "lease": ctx.now + dt.timedelta(seconds=60),
    98	                                "r": run_id,
    99	                            },
   100	                        )
   101	                        outcome = execution.execute(ctx.store.load_run(s, run_id), ctx)
   102	                        assert outcome.status == RunStatus.TASK_CREATED.value, outcome
   103	                        delays.append((ctx.now - scheduled_for).total_seconds())
   104	            minute += 1
   105	
   106	    assert len(delays) == SCHEDULES
   107	    ordered = sorted(delays)
   108	    p95 = ordered[max(0, int(len(ordered) * 0.95) - 1)]
   109	    assert p95 <= defaults.SCHEDULE_START_DELAY_P95_S, (p95, statistics.mean(delays))
   110	    assert max(ordered) <= 60
